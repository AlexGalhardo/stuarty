# The configure flags of the own LGPL FFmpeg build (D-017, ADR-0011), one per line,
# in sections: [common] first, then the section of the target operating system.
# This file is the only place that holds them: scripts/ffmpeg-build.sh reads it, and
# `go run ./internal/tools/ffmpegconf -flags .config/ffmpeg-build/configure-flags.txt
# -allow .config/ffmpeg-build/configure-allowlist.txt` fails the build when a flag is
# not licence safe. No --enable-gpl, no --enable-nonfree, no --enable-libopenh264:
# the Cisco OpenH264 binary is never linked. The libopenh264 encoder is switched on by the
# patch of patches/ (pinned in sources.env), which loads the library at run time with
# dlopen or LoadLibrary from VUKEM_VIDEO_OPENH264_DIR (D-017), so no flag is needed for it.
# Hardware encoders (D-034): --enable-ffnvcodec (NVENC), --enable-libvpl (Intel Quick Sync through the
# MIT oneVPL dispatcher) and --enable-amf (AMD AMF, MIT headers) on linux and windows; the vendor
# runtimes are loaded from the GPU driver at run time, never shipped. macOS keeps VideoToolbox.
[common]
--disable-autodetect
--disable-debug
--disable-doc
--disable-ffplay
--disable-network
--disable-shared
--enable-static
--enable-pic
--enable-stripping
--enable-zlib
--enable-libvpx
--enable-libopus
--enable-libzimg
--disable-libopenh264
--disable-libx264
--disable-libx265

[linux]
--enable-ffnvcodec
--enable-nvenc
--enable-libvpl
--enable-amf

[darwin]
--enable-videotoolbox
--enable-audiotoolbox

[windows]
--enable-ffnvcodec
--enable-nvenc
--enable-libvpl
--enable-amf
--enable-mediafoundation
# mfenc.c of FFmpeg 9.0.2 uses the D3D11 types without guarding them, so MediaFoundation needs d3d11va under --disable-autodetect.
--enable-d3d11va
